```python
# app/models/user.py
from sqlalchemy import Column, String, Boolean, DateTime, JSON
from sqlalchemy.orm import relationship
from app.db.session import Base
import uuid
from datetime import datetime

class User(Base):
    __tablename__ = "users"
    
    id = Column(String(36), primary_key=True, default=lambda: str(uuid.uuid4()))
    employee_id = Column(String(50), unique=True, index=True, comment="工号")
    name = Column(String(100), nullable=False, comment="姓名")
    department = Column(String(100), comment="部门")
    position = Column(String(100), comment="职位")
    email = Column(String(200))
    phone = Column(String(20))
    role = Column(String(50), default="employee", comment="角色: employee/approver/admin")
    permissions = Column(JSON, default=list, comment="权限列表")
    is_active = Column(Boolean, default=True)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
    
    # 关联
    conversations = relationship("Conversation", back_populates="user")
```

```python
# app/models/conversation.py
from sqlalchemy import Column, String, Integer, DateTime, JSON, ForeignKey, Enum
from sqlalchemy.orm import relationship
from app.db.session import Base
import enum

class ConversationStatus(str, enum.Enum):
    ACTIVE = "active"
    COMPLETED = "completed"
    TRANSFERRED = "transferred"  # 转人工

class Conversation(Base):
    __tablename__ = "conversations"
    
    id = Column(String(36), primary_key=True)
    user_id = Column(String(36), ForeignKey("users.id"), index=True)
    status = Column(Enum(ConversationStatus), default=ConversationStatus.ACTIVE)
    intent = Column(String(100), comment="识别的意图")
    business_type = Column(String(100), comment="业务类型")
    context = Column(JSON, default=dict, comment="上下文/槽位数据")
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
    
    # 关联
    user = relationship("User", back_populates="conversations")
    messages = relationship("Message", back_populates="conversation", order_by="Message.created_at")
```

```python
# app/api/v1/chat.py
from fastapi import APIRouter, Depends, WebSocket, WebSocketDisconnect
from pydantic import BaseModel
from typing import Optional, List

router = APIRouter(prefix="/chat", tags=["对话"])

# ============ 请求/响应模型 ============

class ChatRequest(BaseModel):
    """对话请求"""
    conversation_id: Optional[str] = None  # 为空则创建新会话
    message: str                           # 用户消息
    attachments: Optional[List[str]] = None  # 附件URL列表

class ChatResponse(BaseModel):
    """对话响应"""
    conversation_id: str
    message_id: str
    content: str                           # 回复内容
    content_type: str = "text"             # text/card/form
    card_data: Optional[dict] = None       # 卡片数据(表单/确认卡片)
    suggestions: Optional[List[str]] = None  # 推荐操作
    requires_action: bool = False          # 是否需要用户操作
    tool_calls: Optional[List[dict]] = None  # 工具调用记录

class ConversationHistory(BaseModel):
    """会话历史"""
    conversation_id: str
    messages: List[dict]
    total: int

# ============ 接口定义 ============

@router.post("/send", response_model=ChatResponse)
async def send_message(
    request: ChatRequest,
    current_user = Depends(get_current_user)
):
    """
    发送消息
    
    - 创建新会话或继续已有会话
    - 支持文本和附件
    - 返回智能体回复
    """
    pass

@router.get("/history/{conversation_id}", response_model=ConversationHistory)
async def get_conversation_history(
    conversation_id: str,
    limit: int = 50,
    offset: int = 0,
    current_user = Depends(get_current_user)
):
    """获取会话历史"""
    pass

@router.post("/confirm/{task_id}")
async def confirm_action(
    task_id: str,
    confirmed: bool,
    current_user = Depends(get_current_user)
):
    """
    确认高风险操作
    
    - confirmed=True: 确认执行
    - confirmed=False: 取消操作
    """
    pass

@router.post("/transfer/{conversation_id}")
async def transfer_to_human(
    conversation_id: str,
    reason: Optional[str] = None,
    current_user = Depends(get_current_user)
):
    """转人工"""
    pass

# ============ WebSocket ============

@router.websocket("/ws/{token}")
async def websocket_chat(websocket: WebSocket, token: str):
    """
    WebSocket 实时对话
    
    消息格式:
    -> {"type": "message", "content": "...", "conversation_id": "..."}
    <- {"type": "reply", "content": "...", "conversation_id": "..."}
    <- {"type": "typing", "is_typing": true}
    <- {"type": "card", "card_data": {...}}
    """
    await websocket.accept()
    try:
        while True:
            data = await websocket.receive_json()
            # 处理消息并流式返回
            pass
    except WebSocketDisconnect:
        pass
```

```python
# app/core/agent/orchestrator.py
from typing import Optional, Dict, Any, List
from dataclasses import dataclass
from enum import Enum
import structlog

logger = structlog.get_logger()

class AgentState(Enum):
    """智能体状态"""
    IDLE = "idle"
    INTENT_RECOGNIZING = "intent_recognizing"
    SLOT_FILLING = "slot_filling"
    VALIDATING = "validating"
    EXECUTING = "executing"
    CONFIRMING = "confirming"
    COMPLETED = "completed"
    FAILED = "failed"
    TRANSFERRED = "transferred"

@dataclass
class AgentContext:
    """智能体上下文"""
    user_id: str
    conversation_id: str
    state: AgentState
    intent: Optional[str] = None
    business_type: Optional[str] = None
    slots: Dict[str, Any] = None
    rag_results: List[Dict] = None
    tool_calls: List[Dict] = None
    risk_level: str = "low"
    
    def __post_init__(self):
        if self.slots is None:
            self.slots = {}
        if self.rag_results is None:
            self.rag_results = []
        if self.tool_calls is None:
            self.tool_calls = []

class Orchestrator:
    """
    智能体编排器
    
    职责:
    1. 协调意图识别、槽位抽取、RAG检索、工具调用
    2. 管理对话状态流转
    3. 处理异常和兜底
    """
    
    def __init__(
        self,
        intent_recognizer,
        slot_extractor,
        rag_retriever,
        tool_registry,
        rule_engine,
        dialog_manager
    ):
        self.intent_recognizer = intent_recognizer
        self.slot_extractor = slot_extractor
        self.rag_retriever = rag_retriever
        self.tool_registry = tool_registry
        self.rule_engine = rule_engine
        self.dialog_manager = dialog_manager
    
    async def process(
        self,
        user_message: str,
        context: AgentContext,
        attachments: Optional[List[str]] = None
    ) -> Dict[str, Any]:
        """
        处理用户消息的主流程
        """
        try:
            # 1. 意图识别
            logger.info("开始处理", user_id=context.user_id, message=user_message[:50])
            intent_result = await self._recognize_intent(user_message, context)
            context.intent = intent_result["intent"]
            context.business_type = intent_result.get("business_type")
            
            # 2. 判断是否需要 RAG 检索
            if self._needs_rag(context.intent):
                rag_results = await self._retrieve_knowledge(user_message, context)
                context.rag_results = rag_results
            
            # 3. 槽位抽取
            slots = await self._extract_slots(user_message, context, attachments)
            context.slots.update(slots)
            
            # 4. 检查槽位完整性
            missing_slots = self._check_slots_completeness(context)
            
            if missing_slots:
                # 需要多轮澄清
                context.state = AgentState.SLOT_FILLING
                return self._generate_clarification_response(missing_slots, context)
            
            # 5. 规则校验
            context.state = AgentState.VALIDATING
            validation_result = await self._validate_rules(context)
            
            if not validation_result["passed"]:
                return self._generate_validation_error_response(validation_result, context)
            
            # 6. 判断风险等级和是否需要人工确认
            context.risk_level = self._assess_risk_level(context)
            
            if context.risk_level == "high" and not context.get("confirmed"):
                context.state = AgentState.CONFIRMING
                return self._generate_confirmation_request(context)
            
            # 7. 执行工具调用
            context.state = AgentState.EXECUTING
            result = await self._execute_tools(context)
            
            # 8. 生成响应
            context.state = AgentState.COMPLETED
            return self._generate_success_response(result, context)
            
        except Exception as e:
            logger.error("处理异常", error=str(e), context=context)
            return await self._handle_error(e, context)
    
    async def _recognize_intent(
        self, 
        message: str, 
        context: AgentContext
    ) -> Dict:
        """意图识别"""
        return await self.intent_recognizer.recognize(
            message=message,
            conversation_history=await self.dialog_manager.get_history(
                context.conversation_id
            )
        )
    
    def _needs_rag(self, intent: str) -> bool:
        """判断是否需要 RAG 检索"""
        rag_intents = [
            "policy_query",      # 制度查询
            "process_inquiry",   # 流程咨询
            "faq"                # 常见问题
        ]
        return intent in rag_intents
    
    async def _retrieve_knowledge(
        self, 
        query: str, 
        context: AgentContext
    ) -> List[Dict]:
        """RAG 知识检索"""
        return await self.rag_retriever.retrieve(
            query=query,
            top_k=5,
            filters={"category": context.business_type}
        )
    
    async def _extract_slots(
        self, 
        message: str, 
        context: AgentContext,
        attachments: Optional[List[str]] = None
    ) -> Dict:
        """槽位抽取"""
        return await self.slot_extractor.extract(
            message=message,
            intent=context.intent,
            business_type=context.business_type,
            existing_slots=context.slots,
            attachments=attachments
        )
    
    def _check_slots_completeness(self, context: AgentContext) -> List[str]:
        """检查槽位完整性，返回缺失的槽位列表"""
        required_slots = self._get_required_slots(context.business_type)
        return [
            slot for slot in required_slots 
            if slot not in context.slots or context.slots[slot] is None
        ]
    
    def _get_required_slots(self, business_type: str) -> List[str]:
        """获取业务类型所需的必填槽位"""
        slot_mapping = {
            "leave": ["leave_type", "start_date", "end_date"],
            "expense": ["expense_type", "amount", "invoice"],
            "meeting_room": ["date", "time", "capacity", "attendees"],
            "material": ["item_name", "quantity"],
            "travel": ["destination", "start_date", "end_date", "purpose"],
        }
        return slot_mapping.get(business_type, [])
    
    async def _validate_rules(self, context: AgentContext) -> Dict:
        """规则校验"""
        return await self.rule_engine.validate(
            business_type=context.business_type,
            slots=context.slots,
            user_id=context.user_id
        )
    
    def _assess_risk_level(self, context: AgentContext) -> str:
        """评估风险等级"""
        high_risk_types = ["expense", "seal", "contract", "budget"]
        medium_risk_types = ["material", "leave", "travel"]
        
        if context.business_type in high_risk_types:
            return "high"
        elif context.business_type in medium_risk_types:
            return "medium"
        return "low"
    
    async def _execute_tools(self, context: AgentContext) -> Dict:
        """执行工具调用"""
        tool = self.tool_registry.get_tool(context.business_type)
        if not tool:
            raise ValueError(f"未找到工具: {context.business_type}")
        
        result = await tool.execute(context.slots, context.user_id)
        context.tool_calls.append({
            "tool": context.business_type,
            "input": context.slots,
            "output": result
        })
        return result
    
    def _generate_clarification_response(
        self, 
        missing_slots: List[str], 
        context: AgentContext
    ) -> Dict:
        """生成多轮澄清响应"""
        # 根据缺失槽位生成询问话术
        slot_questions = {
            "leave_type": "请问您要请什么假？（年假/调休/事假/病假）",
            "start_date": "请问开始日期是哪天？",
            "end_date": "请问结束日期是哪天？",
            "amount": "请问金额是多少？",
            "invoice": "请上传发票附件",
        }
        
        # 一次只问一个关键项
        question = slot_questions.get(missing_slots[0], f"请提供{missing_slots[0]}信息")
        
        return {
            "content": question,
            "content_type": "text",
            "requires_action": True,
            "state": AgentState.SLOT_FILLING.value
        }
    
    def _generate_confirmation_request(self, context: AgentContext) -> Dict:
        """生成人工确认请求"""
        return {
            "content": "请确认以下操作",
            "content_type": "card",
            "card_data": {
                "type": "confirmation",
                "business_type": context.business_type,
                "data": context.slots,
                "risk_level": context.risk_level,
                "warning": "此操作将产生重要影响，请仔细核对"
            },
            "requires_action": True,
            "state": AgentState.CONFIRMING.value
        }
    
    def _generate_success_response(self, result: Dict, context: AgentContext) -> Dict:
        """生成成功响应"""
        return {
            "content": result.get("message", "操作已完成"),
            "content_type": "text",
            "tool_calls": context.tool_calls,
            "task_id": result.get("task_id"),
            "state": AgentState.COMPLETED.value
        }
    
    async def _handle_error(self, error: Exception, context: AgentContext) -> Dict:
        """异常处理"""
        logger.error("处理失败", error=str(error))
        
        # 判断是否需要转人工
        if isinstance(error, (ToolExecutionError, ExternalSystemError)):
            return {
                "content": "系统暂时无法完成您的请求，已为您转接人工客服",
                "content_type": "text",
                "requires_action": False,
                "transfer_to_human": True,
                "state": AgentState.TRANSFERRED.value
            }
        
        return {
            "content": "抱歉，处理过程中出现错误，请稍后重试",
            "content_type": "text",
            "requires_action": False,
            "state": AgentState.FAILED.value
        }
```

```python
logger = structlog.get_logger()

class IntentRecognizer:

    使用 LLM 进行意图分类和业务类型判断
    """
    
    INTENT_PROMPT = """你是一个企业行政助手的意图识别模块。
    
根据用户的输入，识别用户的意图和业务类型。

## 支持的意图
- policy_query: 制度/规则查询（"年假有几天""报销标准是什么"）
- process_inquiry: 流程咨询（"怎么报销""请假流程是什么"）

- material_request: 物资领用
- travel_request: 差旅申请
- seal_request: 用印/盖章
- certificate_request: 证明开具
- status_query: 状态查询（"我的报销到哪一步了"）
- greeting: 打招呼

## 输出格式
请以JSON格式输出：
{
    "intent": "意图类型",
    "business_type": "业务类型(仅事务类意图需要)",
    "confidence": 0.95,
    "reasoning": "识别依据"
}

## 用户输入
{message}

## 对话历史
{history}
"""
    
    def __init__(self, llm_client: AsyncOpenAI, model: str = "gpt-4o-mini"):
        self.llm = llm_client
        self.model = model
    
    async def recognize(
        self,
        message: str,
        conversation_history: Optional[List[Dict]] = None
    ) -> Dict:
        """
        意图识别
        
        Args:
            message: 用户输入
            conversation_history: 对话历史
            
        Returns:
            意图识别结果
        """
        # 构建历史文本
        history_text = ""
        if conversation_history:
            for msg in conversation_history[-5:]:  # 只取最近5轮
                history_text += f"{msg['role']}: {msg['content']}\n"
        
        prompt = self.INTENT_PROMPT.format(
            message=message,
            history=history_text or "无"
        )
        
        try:
            response = await self.llm.chat.completions.create(
                model=self.model,
                messages=[{"role": "user", "content": prompt}],
                temperature=0,
                response_format={"type": "json_object"}
            )
            
            result = json.loads(response.choices[0].message.content)
            
            logger.info(
                "意图识别完成",
                intent=result.get("intent"),
                confidence=result.get("confidence")
            )
            
            # 低置信度时降级处理
            if result.get("confidence", 0) < 0.7:
                result["intent"] = "other"
                result["needs_transfer"] = True
            
            return result
            
        except Exception as e:
            logger.error("意图识别失败", error=str(e))
            return {
                "intent": "other",
                "confidence": 0,
                "needs_transfer": True,
                "error": str(e)
            }
```

```python
# app/core/rag/retriever.py
from typing import List, Dict, Optional
import chromadb
from chromadb.config import Settings
import structlog

logger = structlog.get_logger()

class RAGRetriever:
    """
    RAG 知识检索器
    
    职责:
    1. 从向量库检索相关知识
    2. 重排序和过滤
    3. 返回带引用的结果
    """
    
    def __init__(
        self,
        chroma_client: chromadb.ClientAPI,
        embedder,
        reranker=None,
        collection_name: str = "knowledge_base"
    ):
        self.client = chroma_client
        self.embedder = embedder
        self.reranker = reranker
        self.collection = self.client.get_or_create_collection(
            name=collection_name,
            metadata={"hnsw:space": "cosine"}
        )
    
    async def retrieve(
        self,
        query: str,
        top_k: int = 5,
        filters: Optional[Dict] = None
    ) -> List[Dict]:
        """
        检索相关知识
        
        Args:
            query: 查询文本
            top_k: 返回数量
            filters: 过滤条件
            
        Returns:
            检索结果列表，包含内容、来源、相关度
        """
        try:
            # 1. 向量化查询
            query_embedding = await self.embedder.embed(query)
            
            # 2. 构建过滤条件
            where = {}
            if filters:
                if filters.get("category"):
                    where["category"] = filters["category"]
            
            # 3. 向量检索
            results = self.collection.query(
                query_embeddings=[query_embedding],
                n_results=top_k * 2,  # 多取一些用于重排序
                where=where if where else None,
                include=["documents", "metadatas", "distances"]
            )
            
            # 4. 构建结果
            retrieved = []
            for i, (doc, meta, dist) in enumerate(zip(
                results["documents"][0],
                results["metadatas"][0],
                results["distances"][0]
            )):
                retrieved.append({
                    "content": doc,
                    "source": meta.get("source", "未知来源"),
                    "category": meta.get("category", ""),
                    "relevance": 1 - dist,  # 转换为相关度
                    "chunk_id": meta.get("chunk_id", i)
                })
            
            # 5. 重排序（可选）
            if self.reranker:
                retrieved = await self.reranker.rerank(query, retrieved)
            
            # 6. 返回 top_k
            final_results = retrieved[:top_k]
            
            logger.info(
                "RAG 检索完成",
                query=query[:50],
                results_count=len(final_results)
            )
            
            return final_results
            
        except Exception as e:
            logger.error("RAG 检索失败", error=str(e))
            return []
    
    async def add_document(
        self,
        doc_id: str,
        chunks: List[Dict],
        metadata: Dict
    ):
        """
        添加文档到知识库
        
        Args:
            doc_id: 文档ID
            chunks: 切片列表 [{"content": "...", "index": 0}, ...]
            metadata: 文档元数据
        """
        try:
            # 批量向量化
            contents = [c["content"] for c in chunks]
            embeddings = await self.embedder.embed_batch(contents)
            
            # 构建数据
            ids = [f"{doc_id}_chunk_{c['index']}" for c in chunks]
            metadatas = [
                {
                    **metadata,
                    "doc_id": doc_id,
                    "chunk_index": c["index"]
                }
                for c in chunks
            ]
            
            # 入库
            self.collection.add(
                ids=ids,
                embeddings=embeddings,
                documents=contents,
                metadatas=metadatas
            )
            
            logger.info("文档入库成功", doc_id=doc_id, chunks_count=len(chunks))
            
        except Exception as e:
            logger.error("文档入库失败", doc_id=doc_id, error=str(e))
            raise
    
    async def delete_document(self, doc_id: str):
        """删除文档"""
        try:
            # 查找该文档的所有切片
            results = self.collection.get(
                where={"doc_id": doc_id},
                include=[]
            )
            
            if results["ids"]:
                self.collection.delete(ids=results["ids"])
                
            logger.info("文档删除成功", doc_id=doc_id)
            
        except Exception as e:
            logger.error("文档删除失败", doc_id=doc_id, error=str(e))
            raise
```

```python
# app/core/tools/base.py
from abc import ABC, abstractmethod
from typing import Dict, Any, Optional
import httpx
import structlog
from datetime import datetime

logger = structlog.get_logger()

class ToolExecutionError(Exception):
    """工具执行异常"""
    pass

class ExternalSystemError(Exception):
    """外部系统异常"""
    pass

class BaseTool(ABC):
    """
    工具基类
    
    所有业务工具继承此类，实现统一接口
    """
    
    def __init__(
        self,
        name: str,
        description: str,
        base_url: str,
        timeout: int = 30,
        max_retries: int = 3
    ):
        self.name = name
        self.description = description
        self.base_url = base_url
        self.timeout = timeout
        self.max_retries = max_retries
        self.client = httpx.AsyncClient(timeout=timeout)
    
    @abstractmethod
    async def execute(
        self, 
        params: Dict[str, Any], 
        user_id: str
    ) -> Dict[str, Any]:
        """
        执行工具调用
        
        Args:
            params: 调用参数
            user_id: 操作用户ID
            
        Returns:
            执行结果
        """
        pass
    
    @abstractmethod
    def get_schema(self) -> Dict:
        """获取工具参数 Schema"""
        pass
    
    async def _request(
        self,
        method: str,
        path: str,
        data: Optional[Dict] = None,
        headers: Optional[Dict] = None
    ) -> Dict:
        """
        统一 HTTP 请求方法
        
        包含: 重试、超时、错误处理、审计日志
        """
        url = f"{self.base_url}{path}"
        last_error = None
        
        for attempt in range(self.max_retries):
            try:
                logger.info(
                    "工具调用开始",
                    tool=self.name,
                    method=method,
                    url=url,
                    attempt=attempt + 1
                )
                
                start_time = datetime.utcnow()
                
                if method.upper() == "GET":
                    response = await self.client.get(url, headers=headers)
                elif method.upper() == "POST":
                    response = await self.client.post(
                        url, json=data, headers=headers
                    )
                elif method.upper() == "PUT":
                    response = await self.client.put(
                        url, json=data, headers=headers
                    )
                else:
                    raise ValueError(f"不支持的 HTTP 方法: {method}")
                
                duration = (datetime.utcnow() - start_time).total_seconds()
                
                # 检查响应
                response.raise_for_status()
                result = response.json()
                
                logger.info(
                    "工具调用成功",
                    tool=self.name,
                    status=response.status_code,
                    duration=duration
                )
                
                return result
                
            except httpx.TimeoutException as e:
                last_error = e
                logger.warning(
                    "工具调用超时",
                    tool=self.name,
                    attempt=attempt + 1
                )
                continue
                
            except httpx.HTTPStatusError as e:
                last_error = e
                logger.error(
                    "工具调用HTTP错误",
                    tool=self.name,
                    status=e.response.status_code,
                    response=e.response.text[:200]
                )
                
                # 5xx 错误重试
                if e.response.status_code >= 500:
                    continue
                # 4xx 错误不重试
                raise ExternalSystemError(
                    f"外部系统错误: {e.response.status_code}"
                )
                
            except Exception as e:
                last_error = e
                logger.error("工具调用异常", tool=self.name, error=str(e))
                raise ToolExecutionError(f"工具执行失败: {str(e)}")
        
        # 重试耗尽
        raise ExternalSystemError(
            f"外部系统调用失败(重试{self.max_retries}次): {str(last_error)}"
        )
```

```python
    async def execute(self, params: Dict[str, Any], user_id: str) -> Dict:
        """执行会议室预定"""
        
        # 1. 查询可用会议室
        available_rooms = await self._request(
            "GET",
            "/api/rooms/available",
            data={
                "date": params["date"],
                "start_time": params["start_time"],
                "end_time": params["end_time"],
                "capacity": params["capacity"]
            }
        )
        
        if not available_rooms.get("data"):
            return {
                "success": False,
                "message": f"抱歉，{params['date']} {params['start_time']}-{params['end_time']} 没有符合条件的空闲会议室"
            }
        
        # 2. 选择最佳会议室（优先匹配设备需求）
        selected_room = self._select_best_room(
            available_rooms["data"],
            params.get("equipment", [])
        )
        
        # 3. 预定
        booking_result = await self._request(
            "POST",
            "/api/bookings",
            data={
                "room_id": selected_room["id"],
                "date": params["date"],
                "start_time": params["start_time"],
                "end_time": params["end_time"],
                "booker_id": user_id,
                "attendees": params.get("attendees", [])
            }
        )
        
        return {
            "success": True,
            "message": f"已为您预定 {selected_room['name']}（{selected_room['location']}）",
            "booking_id": booking_result["data"]["id"],
            "room_name": selected_room["name"],
            "room_location": selected_room["location"],
            "date": params["date"],
            "time": f"{params['start_time']}-{params['end_time']}"
        }
    
    def _select_best_room(self, rooms: list, required_equipment: list) -> Dict:
        """选择最佳会议室"""
        if not required_equipment:
            return rooms[0]
        
        # 按设备匹配度排序
        scored = []
        for room in rooms:
            room_equipment = set(room.get("equipment", []))
            match_score = len(set(required_equipment) & room_equipment)
            scored.append((match_score, room))
        
        scored.sort(key=lambda x: x[0], reverse=True)
        return scored[0][1]
```

```python
            "created_at": datetime.utcnow().isoformat(),
            "state": "active",
            "context": {},
            "messages": []
        }
        
        await self.redis.setex(

        )

        """更新会话"""
        session = await self.get_session(session_id)

            raise ValueError(f"会话不存在: {session_id}")
        
        session.update(updates)
        session["updated_at"] = datetime.utcnow().isoformat()

            f"session:{session_id}",
            self.session_ttl,
            json.dumps(session, ensure_ascii=False)
        )
```

```python
# app/middleware/audit.py
from fastapi import Request, Response
from starlette.middleware.base import BaseHTTPMiddleware
from typing import Callable
import json
import uuid
from datetime import datetime
import structlog

logger = structlog.get_logger()

class AuditMiddleware(BaseHTTPMiddleware):
    """
    审计中间件
    
    记录所有 API 请求和响应
    """
    
    # 不需要审计的路径
    EXCLUDE_PATHS = [
        "/health",
        "/metrics",
        "/docs",
        "/openapi.json"
    ]
    
    async def dispatch(self, request: Request, call_next: Callable) -> Response:
        # 检查是否需要审计
        if request.url.path in self.EXCLUDE_PATHS:
            return await call_next(request)
        
        # 生成审计ID
        audit_id = str(uuid.uuid4())
        request.state.audit_id = audit_id
        
        # 记录请求
        start_time = datetime.utcnow()
        
        # 读取请求体
        body = None
        if request.method in ["POST", "PUT", "PATCH"]:
            try:
                body = await request.body()
                body = json.loads(body) if body else None
            except:
                body = "<binary>"
        
        # 获取用户信息（从认证中间件）
        user_id = getattr(request.state, "user_id", "anonymous")
        
        audit_request = {
            "audit_id": audit_id,
            "user_id": user_id,
            "method": request.method,
            "path": request.url.path,
            "query_params": dict(request.query_params),
            "body": body,
            "ip": request.client.host if request.client else None,
            "user_agent": request.headers.get("user-agent"),
            "timestamp": start_time.isoformat()
        }
        
        # 执行请求
        try:
            response = await call_next(request)
            
            # 计算耗时
            duration = (datetime.utcnow() - start_time).total_seconds()
            
            # 记录响应
            audit_response = {
                "status_code": response.status_code,
                "duration": duration
            }
            
            # 异步写入审计日志（不阻塞响应）
            await self._write_audit_log(audit_request, audit_response)
            
            return response
            
        except Exception as e:
            # 记录异常
            audit_response = {
                "error": str(e),
                "duration": (datetime.utcnow() - start_time).total_seconds()
            }
            await self._write_audit_log(audit_request, audit_response)
            raise

            }
```

### 6.2 Nginx 配置

```nginx
# nginx.conf
events {
    worker_connections 1024;
}

http {
    upstream api {
        least_conn;
        server api:8000;
    }

    server {
        listen 80;
        server_name your-domain.com;
        return 301 https://$server_name$request_uri;
    }

    server {
        listen 443 ssl http2;
        server_name your-domain.com;

        ssl_certificate /etc/nginx/ssl/cert.pem;
        ssl_certificate_key /etc/nginx/ssl/key.pem;

        # API 代理
        location /api/ {
            proxy_pass http://api;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            
            # 超时设置
            proxy_connect_timeout 60s;
            proxy_send_timeout 60s;
            proxy_read_timeout 120s;
        }

        # WebSocket 代理
        location /api/v1/chat/ws/ {
            proxy_pass http://api;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
            proxy_set_header Host $host;
            proxy_read_timeout 86400;
        }

        # 静态文件（如有前端）
        location / {
            root /usr/share/nginx/html;
            try_files $uri $uri/ /index.html;
        }
    }
}
```
