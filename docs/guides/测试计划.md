```
┌─────────────────────────────────────┐
│           E2E 端到端测试            │  ← 5%
```

```python
import pytest
from app.core.agent.intent import IntentRecognizer
```

```python
import pytest

        self, client: AsyncClient, auth_headers

            "/api/v1/chat/send",

            headers=auth_headers
        )

        data = response.json()
        assert data["code"] == 0

            "/api/v1/chat/send",
            json={"message": "我想请假"},
            headers=auth_headers
        )
        conversation_id = resp1.json()["data"]["conversation_id"]

        self, client: AsyncClient, auth_headers

        """测试获取会话历史"""

            "/api/v1/chat/send",
            json={"message": "你好"},
            headers=auth_headers
        )
```

## 五、安全测试

### 5.1 测试清单

| 测试项 | 测试方法 | 预期结果 |
|--------|----------|----------|
| SQL 注入 | 输入特殊字符 | 参数被正确转义 |
| XSS 攻击 | 输入脚本代码 | 内容被转义输出 |
| 越权访问 | 访问他人资源 | 返回 403 |
| 未授权访问 | 不带 Token 请求 | 返回 401 |
| 暴力破解 | 高频登录尝试 | 账号被锁定 |
| 敏感信息泄露 | 检查响应内容 | 无密码/Token 泄露 |

### 5.2 测试示例

```python
# tests/test_security/test_auth.py
class TestSecurity:
    """安全测试"""
    
    async def test_unauthorized_access(self, client):
        """测试未授权访问"""
        response = await client.get("/api/v1/tasks/my")
        assert response.status_code == 401
    
    async def test_access_other_user_resource(
        self, client, user_a_headers, user_b_task
    ):
        """测试越权访问他人资源"""
        response = await client.get(
            f"/api/v1/tasks/{user_b_task['id']}",
            headers=user_a_headers
        )
        assert response.status_code == 403
    
    async def test_sql_injection(self, client, auth_headers):
        """测试 SQL 注入"""
        response = await client.post(
            "/api/v1/chat/send",
            json={"message": "'; DROP TABLE users; --"},
            headers=auth_headers
        )
        # 应该正常返回，而不是报错
        assert response.status_code == 200
    
    async def test_rate_limiting(self, client, auth_headers):
        """测试限流"""
        for _ in range(150):  # 超过限制
            await client.post(
                "/api/v1/chat/send",
                json={"message": "测试"},
                headers=auth_headers
            )
        
        response = await client.post(
            "/api/v1/chat/send",
            json={"message": "测试"},
            headers=auth_headers
        )
        assert response.status_code == 429
```
