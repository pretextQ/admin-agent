### 3.2 导入风格

```python
# ✅ 推荐：使用具体导入
from app.admin_ai.db.models import UserModel
from app.admin_ai.core.config import get_config

# ❌ 避免：使用通配符导入
from app.admin_ai.db.models import *

# ✅ 推荐：使用别名简化长模块名
from app.admin_ai.core.agent.orchestrator import Orchestrator as AgentOrchestrator

from __future__ import annotations
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from app.admin_ai.db.models import UserModel
```

### 3.3 相对导入与绝对导入

```python

from app.admin_ai.db.models import UserModel

# ✅ 同一包内可使用相对导入
from .models import UserModel
from .database import Base

# ❌ 避免：跨包使用相对导入
from ..core.config import get_config  # 不推荐
```

## 4. 命名规范

### 4.1 命名风格一览

| 类型 | 风格 | 示例 |
|------|------|------|
| 包名 | 小写无下划线 | `admin_ai`, `core`, `db` |
| 模块名 | 小写下划线 | `user_service.py`, `database.py` |
| 类名 | 大驼峰 | `UserService`, `DatabaseConfig` |
| 函数名 | 小写下划线 | `get_user_by_id()`, `create_task()` |
| 方法名 | 小写下划线 | `def get_user(self)` |
| 变量名 | 小写下划线 | `user_id`, `max_retry_count` |
| 常量 | 大写下划线 | `MAX_RETRY_COUNT`, `DEFAULT_TIMEOUT` |
| 私有属性 | 下划线前缀 | `self._internal_cache` |
| 私有方法 | 下划线前缀 | `def _validate_input(self)` |
| 类型变量 | 大驼峰 | `T`, `UserType` |
| 数据库表名 | 小写下划线 | `users`, `audit_logs` |
| 数据库列名 | 小写下划线 | `user_id`, `created_at` |

## 9. 异步编程

### 9.1 异步函数

```python
# ✅ IO 操作使用异步
async def get_user_from_db(user_id: str) -> Optional[UserModel]:
    """从数据库获取用户。"""
    async with get_db_session() as session:
        result = await session.execute(
            select(UserModel).where(UserModel.id == user_id)
        )
        return result.scalar_one_or_none()

# ✅ HTTP 请求使用异步
async def call_external_api(url: str) -> dict:
    """调用外部 API。"""
    async with httpx.AsyncClient() as client:
        response = await client.get(url)
        return response.json()

# ✅ 并发执行
async def process_batch(items: list[str]) -> list[dict]:
    """批量处理。"""
    tasks = [process_item(item) for item in items]
    return await asyncio.gather(*tasks)
```

### 9.2 上下文管理器

```python
# ✅ 异步上下文管理器
@asynccontextmanager
async def get_db_session() -> AsyncGenerator[AsyncSession, None]:
    """获取数据库会话。"""
    session = async_session_factory()
    try:
        yield session
        await session.commit()
    except Exception:
        await session.rollback()
        raise
    finally:
        await session.close()

# 使用
async def create_user(data: dict) -> UserModel:
    async with get_db_session() as session:
        user = UserModel(**data)
        session.add(user)
        await session.flush()
        return user
```

### 9.3 异步迭代

```python
# ✅ 异步生成器
async def stream_messages(
    conversation_id: str
) -> AsyncGenerator[MessageModel, None]:
    """流式获取消息。"""
    async with get_db_session() as session:
        result = await session.execute(
            select(MessageModel)
            .where(MessageModel.conversation_id == conversation_id)
            .order_by(MessageModel.created_at)
        )
        for message in result.scalars():
            yield message

# 使用
async def process_conversation(conversation_id: str):
    async for message in stream_messages(conversation_id):
        await process_message(message)
```

## 12. SQLAlchemy 模型

### 12.1 模型定义

```python
# -*- coding: utf-8 -*-
"""数据库模型定义。"""

from __future__ import annotations

import uuid
from datetime import datetime
from typing import Any, Optional

from sqlalchemy import Boolean, DateTime, String, JSON
from sqlalchemy.orm import Mapped, mapped_column, relationship

from app.admin_ai.db.database import Base


def generate_uuid() -> str:
    """生成 UUID。"""
    return str(uuid.uuid4())


class UserModel(Base):
    """用户模型。
    
    存储用户基本信息和认证信息。
    """
    
    __tablename__ = "users"
    
    # 主键
    id: Mapped[str] = mapped_column(
        String(36),
        primary_key=True,
        default=generate_uuid
    )
    
    # 业务字段
    employee_id: Mapped[str] = mapped_column(
        String(50),
        unique=True,
        nullable=False,
        index=True,
        comment="工号"
    )
    name: Mapped[str] = mapped_column(
        String(100),
        nullable=False,
        comment="姓名"
    )
    email: Mapped[Optional[str]] = mapped_column(
        String(200),
        nullable=True,
        comment="邮箱"
    )
    
    # 状态字段
    is_active: Mapped[bool] = mapped_column(
        Boolean,
        default=True,
        nullable=False,
        comment="是否启用"
    )
    
    # 时间字段
    created_at: Mapped[datetime] = mapped_column(
        DateTime,
        default=datetime.utcnow,
        nullable=False,
        comment="创建时间"
    )
    updated_at: Mapped[datetime] = mapped_column(
        DateTime,
        default=datetime.utcnow,
        onupdate=datetime.utcnow,
        nullable=False,
        comment="更新时间"
    )
    
    # JSON 字段
```

## 13. FastAPI 路由

### 13.1 路由定义

```python
# -*- coding: utf-8 -*-
"""对话 API 路由。"""

from __future__ import annotations

from typing import Any, Optional

from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel, Field

from app.admin_ai.api.deps_context import get_api_context, APIContext

router = APIRouter(prefix="/chat", tags=["对话"])


# 请求模型
class ChatRequest(BaseModel):
    """对话请求。"""
    message: str = Field(..., description="用户消息")
    conversation_id: Optional[str] = Field(None, description="会话ID")


# 响应模型
class ChatResponse(BaseModel):
    """对话响应。"""
    conversation_id: str
    message_id: str
    content: str


# 路由处理
@router.post("/send", response_model=ChatResponse)
async def send_message(
    request: ChatRequest,
    context: APIContext = Depends(get_api_context)
) -> ChatResponse:
    """发送消息。
    
    发送消息给智能体，创建或继续会话。
    
    Args:
        request: 请求参数
        context: API 上下文
        
    Returns:
        ChatResponse: 对话响应
    """
    try:
        result = await context.chat_service.send_message(
            message=request.message,
            conversation_id=request.conversation_id
        )
        return ChatResponse(**result)
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
```

```python
# routes.py - 路由汇总
from fastapi import APIRouter
```

## 14. Pydantic Schema

### 14.1 Schema 定义

```python
# -*- coding: utf-8 -*-
"""Pydantic Schema 定义。"""

from __future__ import annotations

from datetime import datetime
from typing import Any, Optional

from pydantic import BaseModel, Field, ConfigDict


# 基础 Schema
class BaseSchema(BaseModel):
    """基础 Schema。"""
    
    model_config = ConfigDict(
        from_attributes=True,
        populate_by_name=True
    )


# 请求 Schema
class UserCreateSchema(BaseSchema):
    """创建用户请求。"""
    
    employee_id: str = Field(..., description="工号", min_length=1, max_length=50)
    name: str = Field(..., description="姓名", min_length=1, max_length=100)
    email: Optional[str] = Field(None, description="邮箱", max_length=200)
    department: Optional[str] = Field(None, description="部门", max_length=100)


# 响应 Schema
class UserResponse(BaseSchema):
    """用户响应。"""
    
    id: str = Field(..., description="用户ID")
    employee_id: str = Field(..., description="工号")
    name: str = Field(..., description="姓名")
    email: Optional[str] = Field(None, description="邮箱")
    is_active: bool = Field(True, description="是否启用")
    created_at: datetime = Field(..., description="创建时间")


# 列表响应
class UserListResponse(BaseSchema):
    """用户列表响应。"""
    
    items: list[UserResponse] = Field(..., description="用户列表")
    total: int = Field(..., description="总数")
    page: int = Field(1, description="当前页")
    page_size: int = Field(20, description="每页数量")
```

### 14.2 Schema 规范

| 规范 | 说明 |
|------|------|
| 命名 | 请求：`XxxCreate/UpdateSchema`，响应：`XxxResponse` |
| 继承 | 所有 Schema 继承 BaseSchema |
| 配置 | 使用 `model_config` 配置 |
| 验证 | 使用 Field 添加验证规则 |
| 文档 | 所有字段必须有 description |
